from PIL import Image, ImageDraw, ImageFilter, ImageEnhance
import numpy as np, random, math, os

W,H,S=1920,150,3
w,h=W*S,H*S
rng=random.Random(888)

def noise_tex(size, scale=40, alpha=30, seed=1):
    rr=np.random.default_rng(seed)
    small=rr.normal(128,35,(max(2,size[1]//scale),max(2,size[0]//scale))).clip(0,255).astype(np.uint8)
    im=Image.fromarray(small,'L').resize(size,Image.Resampling.BICUBIC).filter(ImageFilter.GaussianBlur(scale/8))
    out=Image.new('RGBA',size,(255,255,255,0)); out.putalpha(im.point(lambda x:int((x/255)*alpha)))
    return out

def rgba_gradient(size, top, bottom):
    W0,H0=size
    arr=np.zeros((H0,W0,4),dtype=np.uint8)
    for y in range(H0):
        t=y/max(1,H0-1)
        col=[int(top[i]*(1-t)+bottom[i]*t) for i in range(3)]
        arr[y,:,0]=col[0]; arr[y,:,1]=col[1]; arr[y,:,2]=col[2]; arr[y,:,3]=255
    return Image.fromarray(arr,'RGBA')

def smooth1d(arr, sigma=30):
    radius=max(1,int(sigma*3))
    x=np.arange(-radius,radius+1)
    k=np.exp(-(x**2)/(2*sigma**2)); k/=k.sum()
    pad=np.pad(arr,(radius,radius),mode='edge')
    return np.convolve(pad,k,mode='same')[radius:-radius]

def smooth_profile(seed, base_y, amp, points=20, smooth=80):
    rr=np.random.default_rng(seed)
    anchors=np.linspace(0,w,points)
    vals=base_y*S + rr.normal(0,amp*S,points)
    for k in rr.choice(np.arange(2,points-2), size=max(2,points//6), replace=False):
        vals[k]-=rr.uniform(amp*.8,amp*1.8)*S
    x=np.arange(w)
    p=np.interp(x,anchors,vals)
    return smooth1d(p,sigma=smooth)

def mountain_layer(seed, base_y, amp, top_col, bot_col, blur=0, snowline=None, alpha=255):
    profile=smooth_profile(seed,base_y,amp,points=24,smooth=55)
    mask=np.zeros((h,w),dtype=np.uint8)
    ys=np.clip(profile.astype(int),0,h-1)
    for x,y in enumerate(ys):
        mask[y:,x]=alpha
    fill=rgba_gradient((w,h),top_col,bot_col)
    fill.putalpha(Image.fromarray(mask,'L'))
    if snowline:
        snow=np.zeros((h,w),dtype=np.uint8)
        cutoff=int(snowline*S)
        for x,y in enumerate(ys):
            if y<cutoff:
                end=min(h,cutoff+4*S)
                n=end-y
                snow[y:end,x]=np.linspace(160,0,n).astype(np.uint8)
        snowim=Image.new('RGBA',(w,h),(132,182,214,0)); snowim.putalpha(Image.fromarray(snow,'L'))
        fill=Image.alpha_composite(fill,snowim)
    if blur: fill=fill.filter(ImageFilter.GaussianBlur(blur*S))
    return fill, profile

def glow(base,x,y,r,color=(255,168,73),a=90,blur=8):
    g=Image.new('RGBA',(w,h),(0,0,0,0)); gd=ImageDraw.Draw(g)
    gd.ellipse(((x-r)*S,(y-r)*S,(x+r)*S,(y+r)*S),fill=(*color,a))
    base.alpha_composite(g.filter(ImageFilter.GaussianBlur(blur*S)))
